cmake_minimum_required(VERSION 3.22.1)
project(fetchkit_consumer LANGUAGES C CXX)
find_package(fetchkit 0.1.0 EXACT CONFIG REQUIRED)
add_executable(consumer_c consumer.c)
set_target_properties(consumer_c PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(consumer_c PRIVATE fetchkit::core)
add_executable(consumer_cpp consumer.cpp)
target_link_libraries(consumer_cpp PRIVATE fetchkit::core)
add_library(consumer_shared SHARED consumer_bridge.cpp)
target_link_libraries(consumer_shared PRIVATE fetchkit::core)
