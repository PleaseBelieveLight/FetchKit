cmake_minimum_required(VERSION 3.22.1)
project(fetchkit_consumer LANGUAGES C CXX)
if(DEFINED FETCHKIT_EXPECTED_VERSION)
    string(REGEX MATCH "^[0-9]+[.][0-9]+[.][0-9]+" expected_numeric "${FETCHKIT_EXPECTED_VERSION}")
    if(expected_numeric STREQUAL "")
        message(FATAL_ERROR "Invalid expected FetchKit version: ${FETCHKIT_EXPECTED_VERSION}")
    endif()
    find_package(fetchkit "${expected_numeric}" EXACT CONFIG REQUIRED)
    if(NOT fetchkit_VERSION_FULL STREQUAL FETCHKIT_EXPECTED_VERSION)
        message(FATAL_ERROR "Unexpected FetchKit version: ${fetchkit_VERSION_FULL}")
    endif()
else()
    find_package(fetchkit CONFIG REQUIRED)
endif()
add_executable(consumer_c consumer.c)
set_target_properties(consumer_c PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(consumer_c PRIVATE fetchkit::core)
add_executable(consumer_cpp consumer.cpp)
target_link_libraries(consumer_cpp PRIVATE fetchkit::core)
add_library(consumer_shared SHARED consumer_bridge.cpp)
target_link_libraries(consumer_shared PRIVATE fetchkit::core)
